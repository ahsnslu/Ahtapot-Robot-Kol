import matplotlib.pyplot as plt
import numpy as np

def spiro_test(angle_degrees):
    # --- SETTINGS ---
    N = 25
    start_length = 25
    ratio = 0.95

    x = [0]
    y = [0]

    current_angle = np.radians(90)

    print("--- SEGMENT LENGTH ANALYSIS ---")

    for i in range(N):
        current_length = start_length * (ratio ** i)
        current_angle += np.radians(angle_degrees / N)

        new_x = x[-1] + current_length * np.cos(current_angle)
        new_y = y[-1] + current_length * np.sin(current_angle)

        x.append(new_x)
        y.append(new_y)

        if i == 0 or i == N - 1:
            print(f"Segment {i+1}: Length {current_length:.2f} mm")

    plt.figure(figsize=(6, 6))
    plt.plot(x, y, '-o', linewidth=2, markersize=8, color='purple')

    start_diameter = 50
    for i in range(len(x) - 1):
        diameter = start_diameter * (ratio ** i)
        circle = plt.Circle((x[i], y[i]), diameter / 2, color='purple', alpha=0.1)
        plt.gca().add_patch(circle)

    plt.plot([-20, 20], [0, 0], 'k-', linewidth=3)

    plt.axis('equal')
    plt.grid(True, alpha=0.3)
    plt.title(f"Spiro Robot Simulation\nTotal Bend: {angle_degrees}")
    plt.show()

spiro_test(angle_degrees=540)
